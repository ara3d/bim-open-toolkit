answer = rel.sql@1(t1: spaces.relation, t2: zones.relation, sql: """
SELECT z.ZoneName AS Zone,
       round(sum(CASE WHEN v.MetricId = 'NRC.ENERGY.ANNUAL' THEN v.Value END)
             / sum(CASE WHEN v.MetricId = 'NRC.AREA.FLOOR' THEN v.Value END), 2) AS ZoneEnergyUseIntensity_kWh_per_m2_year,
       count(DISTINCT v.GlobalId) AS Spaces,
       round(sum(CASE WHEN v.MetricId = 'NRC.AREA.FLOOR' THEN v.Value END), 1) AS TotalFloorArea_m2,
       round(sum(CASE WHEN v.MetricId = 'NRC.ENERGY.ANNUAL' THEN v.Value END), 1) AS TotalOperationalEnergy_kWh_per_year
FROM t1 v JOIN t2 z ON z.SpaceGlobalId = v.GlobalId
GROUP BY z.ZoneName
ORDER BY ZoneEnergyUseIntensity_kWh_per_m2_year DESC
""");
