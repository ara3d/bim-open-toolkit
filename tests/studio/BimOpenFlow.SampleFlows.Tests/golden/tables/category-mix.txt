about = view.note@1(text: "Revenue per product category with no SQL: table.join puts each product's price and category on its order, table.derive multiplies Quantity by UnitPrice, and the totals are ranked and charted. Structural leads with 8,880 from 2 orders.");
  // Ok
orders = duck.read@1(path: "{SAMPLES}/orders.csv");
  // Ok  table 8 rows x 5 cols  sha 8c29a3917fc1
  //   columns OrderId:Text, CustomerId:Text, ProductId:Text, Quantity:Integer, OrderDate:Text
  //   [0] "O-1001", "C001", "P-10", 120, "2026-01-15"
  //   [1] "O-1002", "C001", "P-11", 40, "2026-02-03"
  //   [2] "O-1003", "C002", "P-10", 300, "2026-02-14"
products = duck.read@1(path: "{SAMPLES}/products.csv");
  // Ok  table 4 rows x 4 cols  sha aaabaa3212d1
  //   columns ProductId:Text, ProductName:Text, UnitPrice:Number, Category:Text
  //   [0] "P-10", "Stud 2x4", 3.85, "Lumber"
  //   [1] "P-11", "Drywall Sheet", 12.5, "Interior"
  //   [2] "P-12", "Steel Beam", 240, "Structural"
priced = table.join@1(a: orders.table, b: products.table, aKey: "ProductId", mode: "inner");
  // Ok  table 8 rows x 8 cols  sha 2076379efe25
  //   columns OrderId:Text, CustomerId:Text, ProductId:Text, Quantity:Integer, OrderDate:Text, ProductName:Text, UnitPrice:Number, Category:Text
  //   [0] "O-1001", "C001", "P-10", 120, "2026-01-15", "Stud 2x4", 3.85, "Lumber"
  //   [1] "O-1002", "C001", "P-11", 40, "2026-02-03", "Drywall Sheet", 12.5, "Interior"
  //   [2] "O-1003", "C002", "P-10", 300, "2026-02-14", "Stud 2x4", 3.85, "Lumber"
revenue = table.derive@1(table: priced.table, name: "Revenue", expr: "Quantity * UnitPrice");
  // Ok  table 8 rows x 9 cols  sha 4bcfe50f6f42
  //   columns OrderId:Text, CustomerId:Text, ProductId:Text, Quantity:Integer, OrderDate:Text, ProductName:Text, UnitPrice:Number, Category:Text, Revenue:Number
  //   [0] "O-1001", "C001", "P-10", 120, "2026-01-15", "Stud 2x4", 3.85, "Lumber", 462
  //   [1] "O-1002", "C001", "P-11", 40, "2026-02-03", "Drywall Sheet", 12.5, "Interior", 500
  //   [2] "O-1003", "C002", "P-10", 300, "2026-02-14", "Stud 2x4", 3.85, "Lumber", 1155
byCategory = table.aggregate@1(table: revenue.table, groupBy: "Category", aggregates: "sum(Revenue) as TotalRevenue, count(*) as Orders");
  // Ok  table 3 rows x 3 cols  sha 2a81e220b033
  //   columns Category:Text, TotalRevenue:Number, Orders:Integer
  //   [0] "Interior", 4681, 3
  //   [1] "Lumber", 1964, 3
  //   [2] "Structural", 8880, 2
ranked = table.sort@1(table: byCategory.table, A: "TotalRevenue", descendingA: "true");
  // Ok  table 3 rows x 3 cols  sha 4be12883360a
  //   columns Category:Text, TotalRevenue:Number, Orders:Integer
  //   [0] "Structural", 8880, 2
  //   [1] "Interior", 4681, 3
  //   [2] "Lumber", 1964, 3
answer = view.table@1(table: ranked.table, title: "Revenue per category");
  // Ok  table 3 rows x 3 cols  sha 4be12883360a
  //   columns Category:Text, TotalRevenue:Number, Orders:Integer
  //   [0] "Structural", 8880, 2
  //   [1] "Interior", 4681, 3
  //   [2] "Lumber", 1964, 3
chart = chart.bar@1(table: ranked.table, labelColumn: "Category", valueColumns: "TotalRevenue", title: "Revenue per category");
