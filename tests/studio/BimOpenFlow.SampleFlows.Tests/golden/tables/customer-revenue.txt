  //   [1] "C002", "Beaver Builds", "Vancouver", "Contractor"

orders = duck.read@1(path: "{SAMPLES}/orders.csv");
  // Ok  table 8 rows x 5 cols  sha 8c29a3917fc1
  //   columns OrderId:Text, CustomerId:Text, ProductId:Text, Quantity:Integer, OrderDate:Text
  //   [0] "O-1001", "C001", "P-10", 120, "2026-01-15"
  //   [1] "O-1002", "C001", "P-11", 40, "2026-02-03"
  //   [2] "O-1003", "C002", "P-10", 300, "2026-02-14"
products = duck.read@1(path: "{SAMPLES}/products.csv");
  // Ok  table 4 rows x 4 cols  sha aaabaa3212d1
  //   columns ProductId:Text, ProductName:Text, UnitPrice:Number, Category:Text
  //   [0] "P-10", "Stud 2x4", 3.85, "Lumber"
  //   [1] "P-11", "Drywall Sheet", 12.5, "Interior"
  //   [2] "P-12", "Steel Beam", 240, "Structural"
joined = sql.query@1(t1: orders.table, t2: products.table, t3: customers.table, sql: "SELECT c.Name AS Customer, o.Quantity * p.UnitPrice AS Revenue FROM t1 o JOIN t2 p ON o.ProductId = p.ProductId JOIN t3 c ON o.CustomerId = c.CustomerId");
  // Ok  table 8 rows x 2 cols  sha 44fed0cbc3e9
  //   columns Customer:Text, Revenue:Number
  //   [0] "Acme Construction", 462
  //   [1] "Acme Construction", 500
  //   [2] "Beaver Builds", 1155
revenue = table.aggregate@1(table: joined.table, groupBy: "Customer", aggregates: "sum(Revenue) as TotalRevenue, count(Revenue) as Orders");
  // Ok  table 5 rows x 3 cols  sha 3aab8ade6df4
  //   columns Customer:Text, TotalRevenue:Number, Orders:Integer
  //   [0] "Acme Construction", 6242, 3
  //   [1] "Beaver Builds", 4586, 2
  //   [2] "Cedar Design", 3600, 1
top = table.sort@1(table: revenue.table, by: "TotalRevenue desc");
