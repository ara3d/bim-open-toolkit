about = view.note@1(text: """
Role: an accessibility reviewer. Question: which doors of the Duplex are narrower than the 850 mm minimum, and where are they?
Rule (an assumption, not legal advice): a door on an accessible path needs at least 850 mm clear width, as NBC 2020 (National Building Code of Canada) 3.8.3.12 is cited by the NRC door check (samples/nrc/door_verdicts.csv, rule DC-W1, marked illustrative). The 'rule' node holds the number and its source, and every step reads it from there.
What is measured: the model has no Pset_DoorCommon.ClearWidth on any door, so the width checked is the IFC OverallWidth ('doors'), leaf plus frame as exported. Clear width is never larger, so a fail is certain and a pass is provisional until clear width is measured. A door with no width is 'unknown' (PassScore empty, grey in both panes), never a pass or a fail.
How to read it: 'chart' counts doors per 50 mm of width, red bins below 850 mm and green at or above, with an 'unknown' bar for doors with no width; the Vega-Lite spec draws the 850 mm rule as a vertical line. 'doorTable' lists every door with its verdict. The 3D view shares the red-green scale: failing doors are red and every other element is faded to 15 percent ('ghosted').
The verdicts for the Duplex are in samples/viz-analyses/README.md.
""");

rule = table.inline@1(rows: "[{\"RuleId\":\"DC-W1\",\"MinWidth_mm\":850.0,\"Source\":\"NBC 2020 3.8.3.12 as cited by samples/nrc/door_verdicts.csv (illustrative, not legal text)\"}]");

verdicts = sql.query@1(t1: millimetres.table, t2: rule.table, sql: "SELECT d.StepId, d.GlobalId, d.Storey, d.Type, d.Width_mm, r.MinWidth_mm, r.RuleId, CASE WHEN d.Width_mm IS NULL THEN 'unknown' WHEN d.Width_mm >= r.MinWidth_mm THEN 'pass' ELSE 'fail' END AS Verdict, CASE WHEN d.Width_mm IS NULL THEN NULL WHEN d.Width_mm >= r.MinWidth_mm THEN CAST(1 AS DOUBLE) ELSE CAST(0 AS DOUBLE) END AS PassScore FROM t1 d CROSS JOIN t2 r ORDER BY d.Width_mm NULLS LAST, d.StepId");
  // Ok  table 14 rows x 9 cols  sha 0978ac55d4d0
  //   columns StepId:Integer, GlobalId:Text, Storey:Text, Type:Text, Width_mm:Number, MinWidth_mm:Number, RuleId:Text, Verdict:Text, PassScore:Number

histogram = sql.query@1(t1: verdicts.table, t2: rule.table, sql: "WITH known AS (SELECT CAST(floor(Width_mm / 50) * 50 AS BIGINT) AS BinStart_mm FROM t1 WHERE Width_mm IS NOT NULL), bins AS (SELECT range AS BinStart_mm FROM range((SELECT min(BinStart_mm) FROM known), (SELECT max(BinStart_mm) FROM known) + 50, 50)) SELECT CAST(b.BinStart_mm AS VARCHAR) || '-' || CAST(b.BinStart_mm + 50 AS VARCHAR) || ' mm' AS Bin, b.BinStart_mm, b.BinStart_mm + 50 AS BinEnd_mm, (SELECT count(*) FROM known k WHERE k.BinStart_mm = b.BinStart_mm) AS Doors, CASE WHEN b.BinStart_mm >= r.MinWidth_mm THEN CAST(1 AS DOUBLE) WHEN b.BinStart_mm + 50 <= r.MinWidth_mm THEN CAST(0 AS DOUBLE) ELSE CAST(0.5 AS DOUBLE) END AS PassScore, r.MinWidth_mm FROM bins b CROSS JOIN t2 r UNION ALL SELECT 'unknown', NULL, NULL, (SELECT count(*) FROM t1 WHERE Width_mm IS NULL), NULL, (SELECT MinWidth_mm FROM t2) ORDER BY BinStart_mm NULLS LAST");
