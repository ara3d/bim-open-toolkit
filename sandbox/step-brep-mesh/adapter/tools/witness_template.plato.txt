//==
// # StepWitnessesLaws
//
// Layer 3's Part21File witnesses, and the expected values of the adapter
// tests (../adapter/ADAPTER-SPEC.md): for each text fixture under
// ../adapter/fixtures/, the Part21File a correct parser gives, written as a
// literal. Layer 3's types and file laws are the contract
// (step-part21.types.plato, step-part21.laws.plato beside this file); this
// file only builds values and checks them against those laws. TASKS.md chunk
// D reuses the same literals as the inputs of the decode tests. Generated,
// with the fixtures, by ../adapter/tools/gen_adapter.py from
// ../adapter/tools/witness_template.plato.txt: edit those, not this file.
//
// Arena layout rule, used by every witness here and required of the adapter
// (IsSameFile compares arenas element by element): visit the instances in
// ascending id order; within an instance, its records in written order (a
// complex instance's parts as written); within a record, its parameters left
// to right; emit each parameter depth-first in post-order, so a list's items
// (and a typed parameter's inner value) come before the list itself. A
// record's attribute indices are the indices of its top-level parameters.
// The rule follows from Law_ParsePrintParseIsIdentity: PrintStep writes
// instances in id order, so any layout that depends on file order breaks
// the round trip for a file written out of order.
//
// Sections, in order:
//  - Building blocks
//  - The unit cube
//  - Variants of the unit cube
//  - The two-cube assembly
//  - Expected diagnostics
//  - Invariants

library StepWitnessesLaws
{
    //==
    // # Building blocks

    // The header every fixture writes, apart from its description and file
    // name. The author is written 'Andr\X2\00E9\X0\ Lefort' and decodes to
    // "Andr" followed by U+00E9 and " Lefort".
    FixtureHeader(description: String, name: String): StepHeader
        => StepHeader(
            [description],
            "2;1",
            name,
            "2026-10-17T00:00:00",
            ["André Lefort"],
            ["BIM Open Toolkit"],
            "hand-written for the Part21File adapter tests",
            "none",
            "",
            ["AUTOMOTIVE_DESIGN { 1 0 10303 214 1 1 1 1 }"]);

    // The attribute arena indices held by a record.
    P21Indices(xs: Array<Integer>): Array<StepAttributeIndex>
        => xs.Map(x => StepAttributeIndex(x));

    // A simple instance #id = name(attributes).
    P21Simple(id: Integer, name: String, attributes: Array<Integer>): StepInstance
        => StepInstance(StepInstanceId(id), StepInstanceBody.Simple(name, attributes.P21Indices));

    // One part of a complex instance.
    P21Part(name: String, attributes: Array<Integer>): StepSimpleRecord
        => StepSimpleRecord(name, attributes.P21Indices);

    // A complex instance #id = (parts), parts as written.
    P21Complex(id: Integer, parts: Array<StepSimpleRecord>): StepInstance
        => StepInstance(StepInstanceId(id), StepInstanceBody.Complex(parts));

    // A decoded string value.
    P21Str(s: String): StepAttribute
        => StepAttribute.StringValue(s);

    // A real value.
    P21Real(x: Number): StepAttribute
        => StepAttribute.RealValue(x);

    // An integer value.
    P21Int(x: Integer): StepAttribute
        => StepAttribute.IntegerValue(x);

    // An enumeration value, the name between the dots.
    P21Enum(name: String): StepAttribute
        => StepAttribute.EnumValue(name);

    // A reference #id.
    P21Ref(id: Integer): StepAttribute
        => StepAttribute.Reference(StepInstanceId(id));

    // A typed parameter name(inner), inner an arena index.
    P21Typed(name: String, inner: Integer): StepAttribute
        => StepAttribute.Typed(name, StepAttributeIndex(inner));

    // An aggregate of arena indices.
    P21List(items: Array<Integer>): StepAttribute
        => StepAttribute.ListValue(items.P21Indices);

    //==
    // # The unit cube

    // adapter/fixtures/unit-cube.stp: 142 instances, an arena of @CUBE_ARENA_SIZE@ values, laid
    // out by the rule at the top of this file. Ids 1 to 8 are
    // the eight CARTESIAN_POINTs, written in the file as #5 to #8 then #1 to
    // #4; every other instance is written in ascending order.
    Witness_Part21UnitBox(_: Part21File): Part21File
        => Part21File(
            FixtureHeader("unit cube for the Part21File adapter tests", "unit-cube.stp"),
            @CUBE_INSTANCES@,
            @CUBE_ARENA@);

    //==
    // # Variants of the unit cube

    // adapter/fixtures/unit-cube-no-shape.stp: the unit cube without #141
    // ADVANCED_BREP_SHAPE_REPRESENTATION and #142
    // SHAPE_DEFINITION_REPRESENTATION, the two highest ids. By the layout
    // rule its arena is the unit cube's first @NS_ARENA@ values.
    Witness_Part21UnitBoxNoShape(_: Part21File): Part21File
    {
        var box = Part21File.Witness_Part21UnitBox;
        return Part21File(box.Header, box.Instances.Take(140), box.Attributes.Take(@NS_ARENA@));
    }

    // adapter/fixtures/unit-cube-dangling.stp: the unit cube with #65 ORIENTED_EDGE
    // naming #999 where it names #56. Same layout; only the value at index
    // @DANG_INDEX@ (the edge_element of #65) differs.
    Witness_Part21UnitBoxDangling(_: Part21File): Part21File
    {
        var box = Part21File.Witness_Part21UnitBox;
        return Part21File(box.Header, box.Instances,
            box.Attributes.Count.MapRange(i => i == @DANG_INDEX@ ? P21Ref(999) : box.Attributes[i]));
    }

    // adapter/fixtures/unit-cube-truncated.stp: the text stops inside #125
    // CLOSED_SHELL. The prefix is every instance complete before the cut,
    // ids 1 to 124 (the points, written out of order, all come before it).
    // By the layout rule its arena is the unit cube's first @TR_ARENA@ values.
    Witness_Part21UnitBoxTruncatedPrefix(_: Part21File): Part21File
    {
        var box = Part21File.Witness_Part21UnitBox;
        return Part21File(box.Header, box.Instances.Take(124), box.Attributes.Take(@TR_ARENA@));
    }

    //==
    // # The two-cube assembly

    // adapter/fixtures/two-cubes-assembly.stp: the unit cube's 142 instances
    // unchanged, then ids 143 to 162: a second geometric context, the
    // assembly product and its SHAPE_REPRESENTATION with two placements
    // (#145 at the origin, #146 at x = 2 mm), two NEXT_ASSEMBLY_USAGE_OCCURRENCEs
    // of the cube, and for each an ITEM_DEFINED_TRANSFORMATION from the cube's
    // origin #127, a complex SHAPE_REPRESENTATION_RELATIONSHIP with
    // transformation, and a CONTEXT_DEPENDENT_SHAPE_REPRESENTATION. Same
    // layout rule: the cube's arena, then values @CUBE_ARENA_SIZE@ to
    // @ASM_ARENA_SIZE@ - 1 for ids 143 to 162.
    Witness_Part21TwoCubes(_: Part21File): Part21File
    {
        var box = Part21File.Witness_Part21UnitBox;
        return Part21File(
            FixtureHeader("two placements of the unit cube for the Part21File adapter tests", "two-cubes-assembly.stp"),
            box.Instances.Concatenate(@ASM_INSTANCES@),
            box.Attributes.Concatenate(@ASM_ARENA@));
    }

    //==
    // # Expected diagnostics

    // The parse of the truncated fixture: one Error, P21-SYNTAX, at the
    // instance the text stops inside. Its File is the truncated prefix and
    // its Ok is false.
    Witness_Part21UnitBoxTruncatedDiagnostics(_: Diagnostic): Array<Diagnostic>
        => [Diagnose(StepDiagnosticCode.SyntaxError, Provenance.AtInstance(StepInstanceId(125)), Number.Unmeasured)];

    // The parse of the dangling fixture: one Warning,
    // P21-DANGLING-REFERENCE, at the instance holding the reference. Its Ok
    // is true.
    Witness_Part21UnitBoxDanglingDiagnostics(_: Diagnostic): Array<Diagnostic>
        => [Diagnose(StepDiagnosticCode.DanglingReference, Provenance.AtInstance(StepInstanceId(65)), Number.Unmeasured)];

    //==
    // # Invariants

    // Every invariant of spec Part21File (03-step/step.spec.plato) that
    // concerns instance i, over all its partners and record slots.
    InstanceInvariantsHold(f: Part21File, i: Integer): Boolean
    {
        var records = f.RecordsAt(i);
        return Law_InstanceIdsPositive(f, i)
            && Law_InstanceIdsAscending(f, i)
            && f.Instances.Count.Range.All(j => Law_InstanceIdsUnique(f, i, j))
            && records.Count.Range.All(r => records[r].Attributes.Count.Range.All(k => Law_RecordIndicesInRange(f, i, r, k)));
    }

    // Every invariant of spec Part21File, the bounded quantifiers of
    // step.spec.plato written as folds, and Law_HeaderNamesASchema.
    SatisfiesFileInvariants(f: Part21File): Boolean
        => f.Instances.Count.Range.All(i => f.InstanceInvariantsHold(i))
            && f.Attributes.Count.Range.All(a => f.ChildrenOf(StepAttributeIndex(a)).Count.Range.All(k => Law_ChildrenPrecedeParent(f, a, k)))
            && Law_HeaderNamesASchema(f);

    // The number of arena values at which Law_ReferencesResolve fails.
    UnresolvedReferenceCount(f: Part21File): Integer
        => f.Attributes.Count.Range.Map(a => Law_ReferencesResolve(f, a) ? 0 : 1).Reduce(0, (s, x) => s + x);

    // Every witness file satisfies the Part21File invariants and names a
    // schema; every reference resolves except the one planted in the
    // dangling witness. Needs a body for InstancePosition (TASKS.md chunk D).
    Law_WitnessFilesSatisfyInvariants(_: Part21File): Boolean
    {
        var box = Part21File.Witness_Part21UnitBox;
        var noShape = Part21File.Witness_Part21UnitBoxNoShape;
        var dangling = Part21File.Witness_Part21UnitBoxDangling;
        var prefix = Part21File.Witness_Part21UnitBoxTruncatedPrefix;
        var twoCubes = Part21File.Witness_Part21TwoCubes;
        return box.SatisfiesFileInvariants && box.UnresolvedReferenceCount == 0
            && noShape.SatisfiesFileInvariants && noShape.UnresolvedReferenceCount == 0
            && dangling.SatisfiesFileInvariants && dangling.UnresolvedReferenceCount == 1
            && prefix.SatisfiesFileInvariants && prefix.UnresolvedReferenceCount == 0
            && twoCubes.SatisfiesFileInvariants && twoCubes.UnresolvedReferenceCount == 0;
    }
}
